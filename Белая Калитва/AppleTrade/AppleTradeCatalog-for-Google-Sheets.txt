/** Белая Калитва / AppleTrade — supplier catalogue and local sale-price views. */
const BK = {
  endpoint: 'https://api.pricemasterapp.ru/belaya-kalitva/catalog',
  secretProperty: 'BELAYA_ILUBLINO_SNAPSHOT_SECRET',
  maxCatalogAgeMinutes: 30,
  tabs: ['телефоны', 'аксессуары', 'макбуки', 'аймак', 'айпады', 'камеры', 'часы', 'наушники', 'пс', 'дайсон', 'рей бэн'],
};

const BK_MARKUP = {
  'телефоны': {cash: 0.10, cashless: 0.22}, 'макбуки': {cash: 0.10, cashless: 0.22}, 'айпады': {cash: 0.10, cashless: 0.22}, 'аймак': {cash: 0.10, cashless: 0.22}, 'пс': {cash: 0.10, cashless: 0.22},
  'дайсон': {cash: 0.15, cashless: 0.25}, 'рей бэн': {cash: 0.15, cashless: 0.25}, 'часы': {cash: 0.15, cashless: 0.25}, 'наушники': {cash: 0.15, cashless: 0.25}, 'камеры': {cash: 0.15, cashless: 0.25},
};

function onOpen() {
  SpreadsheetApp.getUi().createMenu('AppleTrade')
    .addItem('Открыть каталог', 'bkShowSidebar')
    .addItem('Пересобрать каталог', 'bkInstallAndRefresh')
    .addItem('Включить автообновление 15 мин', 'bkInstallAndRefresh')
    .addToUi();
}
function bkShowSidebar() { SpreadsheetApp.getUi().showSidebar(HtmlService.createHtmlOutputFromFile('AppleTradeCatalogSidebar').setTitle('Каталог AppleTrade')); }

function bkRefreshCatalog() {
  const secret = PropertiesService.getScriptProperties().getProperty(BK.secretProperty);
  if (!secret) throw new Error('Не задан Script Property «' + BK.secretProperty + '»; старые цены сохранены.');
  const response = UrlFetchApp.fetch(BK.endpoint, {headers: {'X-PriceFlow-Secret': secret}, muteHttpExceptions: true});
  if (response.getResponseCode() !== 200) throw new Error('Каталог поставщиков недоступен: HTTP ' + response.getResponseCode());
  const payload = JSON.parse(response.getContentText());
  if (!payload || !payload.categories || !payload.refreshedAt) throw new Error('Неверный формат каталога.');
  const refreshed = new Date(payload.refreshedAt);
  if (isNaN(refreshed.getTime()) || Date.now() - refreshed.getTime() > BK.maxCatalogAgeMinutes * 60 * 1000) throw new Error('Каталог поставщиков устарел; старые цены не записаны.');
  const book = SpreadsheetApp.getActive();
  BK.tabs.forEach(name => {
    const sheet = book.getSheetByName(name); if (!sheet) throw new Error('Нет обязательной вкладки: ' + name);
    bkValidateSheet_(sheet, name);
    const rows = payload.categories[name] || [];
    if (name === 'телефоны') bkWritePhones_(sheet, rows); else bkWriteTitlePrice_(sheet, rows);
  });
  PropertiesService.getScriptProperties().setProperty('BK_LAST_REFRESH', JSON.stringify({
    at: new Date().toISOString(), catalogRefreshedAt: payload.refreshedAt,
    total: Number(payload.total || 0), sources: payload.sources || []
  }));
  book.toast('Загружено позиций: ' + payload.total + '. Источники: ' + payload.sources.join(', '), 'AppleTrade', 8);
  return { total: Number(payload.total || 0), sources: payload.sources || [], catalogRefreshedAt: payload.refreshedAt };
}

function bkGetSetup() {
  const p = PropertiesService.getScriptProperties();
  const hasTrigger = ScriptApp.getProjectTriggers().some(t => t.getHandlerFunction() === 'bkRefreshCatalog');
  let last = null;
  try { last = JSON.parse(p.getProperty('BK_LAST_REFRESH') || 'null'); } catch (error) { last = null; }
  return { hasTrigger: hasTrigger, last: last };
}

function bkInstallAndRefresh() {
  bkInstallTrigger();
  return bkRefreshCatalog();
}

function bkValidateSheet_(sheet, name) {
  bkEnsureSaleColumns_(sheet, name);
  const headers = sheet.getRange(1, 1, 1, Math.max(sheet.getLastColumn(), name === 'телефоны' ? 18 : 5)).getValues()[0]
    .map(value => String(value || '').trim().toLowerCase().replace(/\s+/g, ''));
  const expected = name === 'телефоны'
    ? ['model', 'simconfig', 'memorysize', 'color', 'price', 'нал', 'безнал', 'поставщик', '', 'model', 'simconfig', 'memorysize', 'color', 'ramsize', 'price', 'нал', 'безнал', 'поставщик']
    : ['title', 'price', 'нал', 'безнал', 'поставщик'];
  if (expected.some((value, index) => headers[index] !== value)) {
    throw new Error('Неверная шапка вкладки «' + name + '». Каталог не очищен.');
  }
}

function bkEnsureSaleColumns_(sheet, name) {
  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(value => String(value || '').trim().toLowerCase().replace(/\s+/g, ''));
  if (name !== 'телефоны') {
    if (headers.slice(0, 5).join('|') === 'title|price|нал|безнал|поставщик') return;
    if (headers.slice(0, 2).join('|') !== 'title|price') throw new Error('Неверная шапка вкладки «' + name + '».');
    sheet.insertColumnsAfter(2, 3); sheet.getRange(1, 3, 1, 3).setValues([['Нал', 'Безнал', 'Поставщик']]); return;
  }
  const legacy = ['model', 'simconfig', 'memorysize', 'color', 'price', '', 'model', 'simconfig', 'memorysize', 'color', 'ramsize', 'price'];
  const ready = ['model', 'simconfig', 'memorysize', 'color', 'price', 'нал', 'безнал', 'поставщик', '', 'model', 'simconfig', 'memorysize', 'color', 'ramsize', 'price', 'нал', 'безнал', 'поставщик'];
  if (headers.slice(0, ready.length).join('|') === ready.join('|')) return;
  if (headers.slice(0, legacy.length).join('|') !== legacy.join('|')) throw new Error('Неверная шапка вкладки «телефоны».');
  sheet.insertColumnsAfter(5, 3); sheet.getRange(1, 6, 1, 3).setValues([['Нал', 'Безнал', 'Поставщик']]);
  sheet.insertColumnsAfter(15, 3); sheet.getRange(1, 16, 1, 3).setValues([['Нал', 'Безнал', 'Поставщик']]);
}

function bkPrices_(category, price) {
  const raw = Number(price), markup = BK_MARKUP[category];
  return {raw: raw, cash: markup ? Math.round(raw * (1 + markup.cash)) : '', cashless: markup ? Math.round(raw * (1 + markup.cashless)) : ''};
}

function bkWriteTitlePrice_(sheet, rows) {
  const last = Math.max(sheet.getLastRow(), 2); sheet.getRange(2, 1, last - 1, Math.max(sheet.getLastColumn(), 5)).clearContent();
  if (rows.length) sheet.getRange(2, 1, rows.length, 5).setValues(rows.map(r => { const prices = bkPrices_(sheet.getName(), r.price); return [r.title, prices.raw, prices.cash, prices.cashless, r.source || '']; }));
}

function bkWritePhones_(sheet, rows) {
  const last = Math.max(sheet.getLastRow(), 2); sheet.getRange(2, 1, last - 1, Math.max(sheet.getLastColumn(), 18)).clearContent();
  const apple = [], android = [];
  rows.forEach(r => { const parsed = bkPhone_(r.title); if (parsed.apple && parsed.sim === 'Не знаю') return; (parsed.apple ? apple : android).push(bkPhoneValues_(parsed, Number(r.price), r.source || '')); });
  if (apple.length) sheet.getRange(2, 1, apple.length, 8).setValues(apple);
  if (android.length) sheet.getRange(2, 10, android.length, 9).setValues(android);
}

function bkPhoneRow_(title, price, source) {
  const parsed = bkPhone_(title);
  return bkPhoneValues_(parsed, price, source || '');
}
function bkPhoneValues_(parsed, price, source) { const prices = bkPrices_('телефоны', price); return parsed.apple ? [parsed.model, parsed.sim, parsed.memory, parsed.color, prices.raw, prices.cash, prices.cashless, source] : [parsed.model, parsed.sim, parsed.memory, parsed.color, parsed.ram, prices.raw, prices.cash, prices.cashless, source]; }

/**
 * Supplier-catalog parser only. It preserves a status such as `(Asis+)` or
 * `(Asis запак)`, accepts iPhone Air, and never guesses a missing technical
 * characteristic. It is not Avito matching or a markup rule.
 */
function bkPhone_(value) {
  const source = String(value || '').replace(/[\u{1F1E6}-\u{1F1FF}]{2}/gu, ' ').replace(/\s+/g, ' ').trim();
  const conditionMatch = source.match(/\(\s*(asis\b[^)]*)\)/i);
  const condition = conditionMatch ? '(' + conditionMatch[1].trim().replace(/^a/i, 'A') + ') ' : '';
  const text = source.replace(/\(\s*asis\b[^)]*\)/ig, ' ').replace(/\s+/g, ' ').trim();
  const specs = text.match(/\b(\d{1,2})\s*\/\s*(64|128|256|512|1024|2048|1|2)\s*(гб|gb|тб|tb)?\b/i);
  const memoryMatch = specs ? null : text.match(/\b(1|2|64|128|256|512|1024|2048)\s*(гб|gb|тб|tb)\b/i);
  const unit = (number, suffix) => String(number) + ' ' + String(suffix || 'GB').toUpperCase().replace('GB', 'ГБ').replace('TB', 'ТБ');
  const memory = specs ? unit(specs[2], specs[3]) : memoryMatch ? unit(memoryMatch[1], memoryMatch[2]) : '';
  const ram = specs ? unit(specs[1], 'GB') : '';
  const sim = /\b(?:2\s*(?:sim|сим)|dual\s*-?\s*sim)\b/i.test(text) ? '2 SIM' : /(?:1\s*)?sim\s*\+\s*e\s*-?sim/i.test(text) ? 'SIM + eSIM' : /e\s*-?sim/i.test(text) ? 'eSIM' : /\bsim\b/i.test(text) ? 'SIM' : 'Не знаю';
  const color = bkPhoneColor_(text);
  const apple = /\biphone\b/i.test(text);
  const appleModel = text.match(/\biPhone\s+(?:Air|\d+(?:e)?(?:\s+(?:Pro\s+Max|Pro|Plus|Mini|Air))?)\b/i);
  const androidModel = text.match(/\b(?:(?:Samsung\s+)?Galaxy\s+(?:S|A|Z|M)\d+(?:\s+(?:Ultra|FE|Plus))?|(?:Samsung|Xiaomi|Redmi|Poco|Honor|Huawei|OnePlus|Realme|Oppo|Vivo|Asus)\s+[A-Za-z0-9][A-Za-z0-9+\-]*(?:\s+(?:Pro|Ultra|Plus|FE|Lite|Max|Note))?)/i);
  // A source may use an unfamiliar Android model name. Keep it verbatim
  // rather than drop the actual supplier item.
  const model = condition + (apple ? (appleModel ? appleModel[0] : text) : (androidModel ? androidModel[0] : text));
  return { apple:apple, model:model.replace(/\s+/g, ' ').trim(), sim:sim, memory:memory, color:color, ram:ram };
}

function bkPhoneColor_(text) {
  const aliases = [[/\b(?:blue|голубой)\b/i, 'голубой'], [/\b(?:red|красный)\b/i, 'красный'], [/\b(?:silver|серебристый)\b/i, 'серебристый'], [/\b(?:black|черный|чёрный)\b/i, 'черный']];
  const match = aliases.find(pair => pair[0].test(text)); if (match) return match[1];
  const colors = /\b(white|green|pink|purple|yellow|gray|grey|gold|orange|sage|teal|ultramarine|natural|desert|lavander|lavender|cloud\s+white|sky\s+blue|белый|синий|зеленый|розовый|фиолетовый|желтый|серый|золотистый)\b/i;
  return (text.match(colors) || [''])[0];
}

function bkInstallTrigger() {
  ScriptApp.getProjectTriggers().filter(t => t.getHandlerFunction() === 'bkRefreshCatalog').forEach(t => ScriptApp.deleteTrigger(t));
  ScriptApp.newTrigger('bkRefreshCatalog').timeBased().everyMinutes(15).create();
}
